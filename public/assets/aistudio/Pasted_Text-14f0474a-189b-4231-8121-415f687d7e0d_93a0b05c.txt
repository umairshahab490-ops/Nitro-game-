function updateTraffic(dt,interact){
  var pz=G.position+PLAYERZ,bucket=((pnow()/250)|0)&3;
  for(var i=0;i<cars.length;i++){var c=cars[i];
    c.z=(c.z+c.speed*dt)%TRACKLEN;
    if(c.type==='bike')c.offset=c.baseOff+Math.sin(pnow()*0.0005+c.phase)*0.18;
    if((i&3)===bucket){
      for(var j=0;j<cars.length;j++){if(j===i)continue;var o=cars[j];
        var dz=(o.z-c.z+TRACKLEN)%TRACKLEN;
        if(dz<1200&&Math.abs(o.offset-c.offset)<0.3&&o.speed<c.speed)c.speed=o.speed*0.96;}
    }
    if(!interact)continue;
    var rel=(c.z-pz+TRACKLEN)%TRACKLEN;
    if(G.invuln<=0&&G.speed>=c.speed*0.9&&(rel<230||rel>TRACKLEN-140)){
      if(Math.abs(G.playerX-c.offset)<(0.14+c.w)*0.45)crash();
    }
    if(c.prevRel<1500&&rel>TRACKLEN-1500&&G.speed>c.speed){
      var gap=Math.abs(G.playerX-c.offset);
      if(gap<0.55&&G.speed>MAXSP*0.55)nearMiss();
    }
    c.prevRel=rel;
  }
}

  if(Nn&&!G.nitroOn&&G.nitro<=0)sfx.deny();
  G.nitroOn=wantNitro;

  G.offroad=Math.abs(G.playerX)>1.0;
  var target=G.nitroOn?MAXSP*1.32:MAXSP,rate=0.35;
  if(G.nitroOn)rate=0.85;

  var pSeg=findSeg(G.position+PLAYERZ);
  G.playerX+=steer*dt*2.2*(0.3+0.7*pct)*grip;
  G.playerX-=dt*pct*pct*pSeg.curve*cfg.centrif/grip;

var last=pnow(),acc=0,raf=window.requestAnimationFrame||function(f){setTimeout(function(){f(pnow());},16);};
function frame(t){
  var dt=clamp((t-last)/1000,0,0.1);last=t;
  var wrap=$('wrap');
  if(wrap.clientWidth!==W||wrap.clientHeight!==H)resize();
  if(G.state!=='PAUSED'){
    acc+=dt;var steps=0;
    while(acc>=STEP&&steps<4){update(STEP);acc-=STEP;steps++;}
    if(steps===4)acc=0;
    render(dt);
  }
  raf(frame);
}

function tap(el,fn){
  var lastT=0;
  var h=function(e){var t=Date.now();if(t-lastT<350)return;lastT=t;fn(e);};
  if(window.PointerEvent)el.addEventListener('pointerdown',h);
  else{el.addEventListener('touchstart',h,{passive:true});el.addEventListener('mousedown',h);}
  el.addEventListener('click',h);
}
var KEYMAP={ArrowLeft:'l',KeyA:'l',ArrowRight:'r',KeyD:'r',ArrowDown:'b',KeyS:'b',ControlLeft:'b',
  Space:'n',ShiftLeft:'n',KeyW:'n',ArrowUp:'n'};
window.addEventListener('keydown',function(e){
  if(KEYMAP[e.code]){keys[KEYMAP[e.code]]=true;e.preventDefault();}
  if(e.code==='KeyP'||e.code==='Escape'){if(G.state==='PAUSED')resumeGame();else pauseGame();}
  if(e.code==='KeyM')setMute(!A.muted);
  if(e.code==='Enter'&&(G.state==='MENU'||G.state==='OVER'))initRun();
});
window.addEventListener('keyup',function(e){if(KEYMAP[e.code]){keys[KEYMAP[e.code]]=false;e.preventDefault();}});
document.addEventListener('contextmenu',function(e){e.preventDefault();});
document.addEventListener('visibilitychange',function(){if(document.hidden)pauseGame();});

function resize(){
  var wrap=$('wrap');
  W=Math.max(1,wrap.clientWidth);H=Math.max(1,wrap.clientHeight);
  DPR=Math.min(2,window.devicePixelRatio||1);
  cv.width=Math.round(W*DPR);cv.height=Math.round(H*DPR);
  ctx.setTransform(DPR,0,0,DPR,0,0);
  vign=ctx.createRadialGradient(W/2,H*0.45,Math.min(W,H)*0.42,W/2,H*0.5,Math.max(W,H)*0.78);
  vign.addColorStop(0,'rgba(0,0,0,0)');vign.addColorStop(1,'rgba(0,0,0,0.42)');
  nitroGlow=ctx.createRadialGradient(W/2,H*0.5,Math.min(W,H)*0.3,W/2,H*0.5,Math.max(W,H)*0.72);
  nitroGlow.addColorStop(0,'rgba(34,230,255,0)');nitroGlow.addColorStop(1,'rgba(34,230,255,0.5)');
}
